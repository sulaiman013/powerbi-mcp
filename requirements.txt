# Power BI MCP Server dependencies (full install: live connectivity on Windows)
#
# These are minimum floors. For a reproducible production deploy, generate a pinned, hashed
# lock file from this file (e.g. `pip-compile --generate-hashes`) and install from that instead.

# MCP SDK. >=1.10.0 is required for tool outputSchema + structuredContent (1.9 lacks them);
# <2 because mcp 2.0 replaced the low-level Server API this server is built on (the server
# refuses to start on 2.x with a message saying so).
mcp>=1.10.0,<2

# Power BI Service: service-principal sign-in and REST
msal>=1.24.0
requests>=2.31.0
python-dotenv>=1.0.0

# Power BI Desktop instance discovery. 6.0+ for Process.net_connections().
psutil>=6.0.0

# Analysis Services (Desktop, XMLA) and live model writes (TOM). pythonnet hosts the ADOMD.NET
# and AMO/TOM assemblies, which must be installed separately on Windows (see README).
pythonnet>=3.0.0

# Access policies (config/policies.yaml)
pyyaml>=6.0.0
